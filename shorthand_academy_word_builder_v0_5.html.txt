        const vowels = [
            { name: 'AH', symbol: '•', position: 1, type: 'long' },

        let currentMode = null;
        let currentWordIndex = 0;
        let currentStep = 1;
        let userBuild = [];
        let selectedQuizOption = null;
        let stats = { wordsCompleted: 0, correctAttempts: 0, totalAttempts: 0, streak: 0, timeSpent: 0, masteredWords: [] };

        function init() { loadStats(); updateStatsDisplay(); }
        function showScreen(screenId) { document.querySelectorAll('.card').forEach(card => { if (card.id !== screenId) card.classList.add('hidden'); }); document.getElementById(screenId).classList.remove('hidden'); }
        function startMode(mode) { currentMode = mode; if (mode === 'guided') { document.getElementById('welcomeScreen').classList.add('hidden'); document.getElementById('guidedScreen').classList.remove('hidden'); loadWord(0); } else if (mode === 'quiz') { document.getElementById('welcomeScreen').classList.add('hidden'); document.getElementById('quizScreen').classList.remove('hidden'); loadQuizWord(); } else if (mode === 'practice') { document.getElementById('welcomeScreen').classList.add('hidden'); document.getElementById('practiceScreen').classList.remove('hidden'); loadPracticeList(); } else if (mode === 'stats') { document.getElementById('welcomeScreen').classList.add('hidden'); document.getElementById('statsScreen').classList.remove('hidden'); updateStatsDisplay(); } }

        function loadWord(index) {
            if (index >= wordDatabase.length) index = 0;
            currentWordIndex = index;
            const word = wordDatabase[index];
            document.getElementById('currentWord').textContent = word.word;
            document.getElementById('wordPronunciation').textContent = `${word.pronunciation} - as in "${word.example}"`;
            const breakdown = document.getElementById('soundBreakdown');
            breakdown.innerHTML = '';
            word.sounds.forEach((sound, i) => {
                const chip = document.createElement('div');
                chip.className = `sound-chip ${['A', 'E', 'I', 'O', 'U', 'AH', 'AW', 'OO'].includes(sound) ? 'vowel' : ''}`;
                chip.textContent = sound;
                breakdown.appendChild(chip);
            });
            currentStep = 1;
            userBuild = [];
            updateStepDisplay();
            setupConsonantOptions(1);
        }

        function updateStepDisplay() {
            for (let i = 1; i <= 4; i++) {
                const step = document.getElementById(`step${i}`);
                const content = document.getElementById(`step${i}Content`);
                if (i < currentStep) { step.className = 'step completed'; content.classList.add('hidden'); }
                else if (i === currentStep) { step.className = 'step active'; content.classList.remove('hidden'); }
                else { step.className = 'step'; content.classList.add('hidden'); }
            }
            document.getElementById('checkBtn').classList.toggle('hidden', currentStep === 4);
        }

        function setupConsonantOptions(stepNum) {
            const container = document.getElementById(`consonantOptions${stepNum}`);
            container.innerHTML = '';
            consonants.forEach(c => {
                const btn = document.createElement('div');
                btn.className = 'consonant-btn';
                btn.textContent = c;
                btn.onclick = () => selectConsonant(c, stepNum, btn);
                container.appendChild(btn);
            });
        }

        function setupVowelOptions() {
            const container = document.getElementById('vowelOptions');
            container.innerHTML = '';
            vowels.forEach(v => {
                const btn = document.createElement('div');
                btn.className = 'vowel-btn';
                btn.textContent = v.name;
                btn.onclick = () => selectVowel(v, btn);
                container.appendChild(btn);
            });
        }

        function selectConsonant(consonant, stepNum, btnElement) {
            const container = document.getElementById(`consonantOptions${stepNum}`);
            container.querySelectorAll('.consonant-btn').forEach(b => b.classList.remove('selected'));
            btnElement.classList.add('selected');
            if (stepNum === 1) userBuild[0] = consonant;
            else userBuild[2] = consonant;
        }

        function selectVowel(vowel, btnElement) {
            document.getElementById('vowelOptions').querySelectorAll('.vowel-btn').forEach(b => b.classList.remove('selected'));
            btnElement.classList.add('selected');
            userBuild[1] = vowel.name;
        }

        function checkCurrentStep() {
            const word = wordDatabase[currentWordIndex];
            const feedback = document.getElementById('buildFeedback');
            feedback.classList.remove('hidden', 'correct', 'incorrect');
            stats.totalAttempts++;

            if (currentStep === 1) {
                if (userBuild[0] === word.sounds[0]) {
                    feedback.classList.add('correct');
                    feedback.textContent = '✓ Correct! First consonant is right.';
                    setTimeout(() => { currentStep = 2; userBuild = [userBuild[0]]; updateStepDisplay(); setupVowelOptions(); feedback.classList.add('hidden'); }, 1000);
                } else {
                    feedback.classList.add('incorrect');
                    feedback.textContent = `✗ Not quite. The first sound in "${word.word}" is ${word.sounds[0]}. Try again!`;
                }
            } else if (currentStep === 2) {
                if (userBuild[1] === word.sounds[1]) {
                    feedback.classList.add('correct');
                    feedback.textContent = '✓ Perfect! Vowel is correct.';
                    setTimeout(() => { currentStep = 3; userBuild = [userBuild[0], userBuild[1]]; updateStepDisplay(); setupConsonantOptions(2); feedback.classList.add('hidden'); }, 1000);
                } else {
                    feedback.classList.add('incorrect');
                    feedback.textContent = `✗ The vowel sound in "${word.word}" is ${word.sounds[1]}. Listen again!`;
                }
            } else if (currentStep === 3) {
                if (userBuild[2] === word.sounds[2]) {
                    feedback.classList.add('correct');
                    feedback.textContent = '✓ Excellent! All sounds are correct.';
                    setTimeout(() => { currentStep = 4; updateStepDisplay(); showOutline(); feedback.classList.add('hidden'); }, 1000);
                } else {
                    feedback.classList.add('incorrect');
                    feedback.textContent = `✗ The last sound in "${word.word}" is ${word.sounds[2]}. Try again!`;
                }
            }
            saveStats();
        }

        function showOutline() {
            const word = wordDatabase[currentWordIndex];
            const outline = userBuild.join('-');
            document.getElementById('outlineDisplay').textContent = outline;
            document.getElementById('outlineFeedback').textContent = `💡 In Pitman shorthand, "${word.word}" is written: ${outline}`;
        }

        function nextWord() { currentWordIndex++; if (currentWordIndex >= wordDatabase.length) currentWordIndex = 0; loadWord(currentWordIndex); }

        function loadQuizWord() {
            const index = Math.floor(Math.random() * wordDatabase.length);
            const word = wordDatabase[index];
            document.getElementById('quizWord').textContent = word.word;
            const options = document.getElementById('quizOptions');
            options.innerHTML = '';
            let quizWords = [word];
            while (quizWords.length < 4) {
                const random = wordDatabase[Math.floor(Math.random() * wordDatabase.length)];
                if (!quizWords.includes(random)) quizWords.push(random);
            }
            quizWords.sort(() => Math.random() - 0.5);
            quizWords.forEach((w, idx) => {
                const option = document.createElement('div');
                option.className = 'quiz-outline-option';
                option.textContent = w.outline;
                option.onclick = () => selectQuizOption(idx, w.word);
                options.appendChild(option);
            });
            selectedQuizOption = null;
            document.getElementById('quizFeedback').classList.add('hidden');
        }

        function selectQuizOption(index, word) {
            document.querySelectorAll('.quiz-outline-option').forEach((opt, idx) => { opt.classList.remove('selected'); if (idx === index) opt.classList.add('selected'); });
            selectedQuizOption = word;
        }

        function checkQuizAnswer() {
            if (!selectedQuizOption) { alert('Please select an answer!'); return; }
            const word = wordDatabase.find(w => w.word === document.getElementById('quizWord').textContent);
            const feedback = document.getElementById('quizFeedback');
            feedback.classList.remove('hidden', 'correct', 'incorrect');
            stats.totalAttempts++;

            if (selectedQuizOption === word.word) {
                feedback.classList.add('correct');
                feedback.textContent = '✓ Correct! Well done!';
                stats.correctAttempts++;
                stats.wordsCompleted++;
                if (!stats.masteredWords.includes(word.word)) stats.masteredWords.push(word.word);
                document.querySelectorAll('.quiz-outline-option').forEach(opt => { if (opt.textContent === word.outline) opt.classList.add('correct'); });
                saveStats();
                setTimeout(() => { loadQuizWord(); }, 1500);
            } else {
                feedback.classList.add('incorrect');
                feedback.textContent = `✗ Not quite. The correct outline for "${word.word}" is ${word.outline}`;
            }
            saveStats();
        }

        function loadPracticeList() {
            const list = document.getElementById('practiceWordList');
            list.innerHTML = '';
            wordDatabase.forEach((word, idx) => {
                const item = document.createElement('div');
                item.className = 'word-item';
                if (stats.masteredWords.includes(word.word)) item.classList.add('completed');
                item.innerHTML = `<div class="word-text">${word.word}</div><div class="word-sounds">${word.sounds.join(' - ')}</div>`;
                item.onclick = () => { currentWordIndex = idx; currentMode = 'guided'; document.getElementById('practiceScreen').classList.add('hidden'); document.getElementById('guidedScreen').classList.remove('hidden'); loadWord(idx); };
                list.appendChild(item);
            });
        }

        function updateStatsDisplay() {
            document.getElementById('wordsCompleted').textContent = stats.wordsCompleted;
            const accuracy = stats.totalAttempts > 0 ? Math.round((stats.correctAttempts / stats.totalAttempts) * 100) : 0;
            document.getElementById('accuracy').textContent = `${accuracy}%`;
            document.getElementById('streak').textContent = stats.streak;
            document.getElementById('timeSpent').textContent = stats.timeSpent;
            const masteredList = document.getElementById('masteredWords');
            if (masteredList) {
                masteredList.innerHTML = '';
                stats.masteredWords.forEach(word => {
                    const item = document.createElement('div');
                    item.className = 'word-item completed';
                    item.innerHTML = `<div class="word-text">${word}</div>`;
                    masteredList.appendChild(item);
                });
            }
            const percentage = (stats.wordsCompleted / wordDatabase.length) * 100;
            document.getElementById('progressFill').style.width = `${percentage}%`;
            document.getElementById('progressText').textContent = `${stats.wordsCompleted} / ${wordDatabase.length} words completed`;
        }

        function playQuizWordSound() { const word = document.getElementById('quizWord').textContent; const wordData = wordDatabase.find(w => w.word === word); alert(`🔊 Playing: "${word}" (${wordData.pronunciation})`); }
        function backToWelcome() { document.querySelectorAll('.card').forEach(card => card.classList.add('hidden')); document.getElementById('welcomeScreen').classList.remove('hidden'); }

        function saveStats() { localStorage.setItem('shorthandWordBuilderStats', JSON.stringify(stats)); updateStatsDisplay(); }
        function loadStats() { const saved = localStorage.getItem('shorthandWordBuilderStats'); if (saved) stats = JSON.parse(saved); }
        function resetProgress() { if (confirm('Are you sure you want to reset all progress?')) { stats = { wordsCompleted: 0, correctAttempts: 0, totalAttempts: 0, streak: 0, timeSpent: 0, masteredWords: [] }; saveStats(); updateStatsDisplay(); } }
