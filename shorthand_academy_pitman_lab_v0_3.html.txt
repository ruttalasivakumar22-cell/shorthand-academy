    <script>
        const consonants = [
            { letter: 'P', name: 'P', type: 'light', direction: 'Downward', example: 'pat, spin, tap', description: 'Light downward stroke' },
            { letter: 'B', name: 'B', type: 'heavy', direction: 'Downward', example: 'bat, obey, tub', description: 'Heavy downward stroke' },
            { letter: 'T', name: 'T', type: 'light', direction: 'Downward', example: 'top, stick, net', description: 'Light downward stroke' },
            { letter: 'D', name: 'D', type: 'heavy', direction: 'Downward', example: 'dog, adobe, rod', description: 'Heavy downward stroke' },
            { letter: 'CH', name: 'CH', type: 'light', direction: 'Downward', example: 'chat, nature, much', description: 'Light downward stroke' },
            { letter: 'J', name: 'J', type: 'heavy', direction: 'Downward', example: 'jam, age, edge', description: 'Heavy downward stroke' },
            { letter: 'K', name: 'K', type: 'light', direction: 'Horizontal', example: 'cat, key, back', description: 'Light horizontal stroke' },
            { letter: 'G', name: 'G', type: 'heavy', direction: 'Horizontal', example: 'go, big, egg', description: 'Heavy horizontal stroke' }
        ];

        let currentLesson = 0;
        let completedLessons = 0;
        let selectedQuizOption = null;
        let practiceCanvas, practiceCtx, isDrawing = false;

        function init() {
            practiceCanvas = document.getElementById('practiceCanvas');
            practiceCtx = practiceCanvas.getContext('2d');
            setupCanvas();
            loadProgress();
            displayConsonantGrid();
        }

        function setupCanvas() {
            practiceCtx.strokeStyle = '#667eea';
            practiceCtx.lineWidth = 3;
            practiceCtx.lineCap = 'round';
            practiceCtx.lineJoin = 'round';
            practiceCanvas.addEventListener('mousedown', startDrawing);
            practiceCanvas.addEventListener('mousemove', draw);
            practiceCanvas.addEventListener('mouseup', stopDrawing);
            practiceCanvas.addEventListener('mouseout', stopDrawing);
            practiceCanvas.addEventListener('touchstart', (e) => { e.preventDefault(); startDrawing(e.touches[0]); });
            practiceCanvas.addEventListener('touchmove', (e) => { e.preventDefault(); draw(e.touches[0]); });
            practiceCanvas.addEventListener('touchend', stopDrawing);
        }

        function startDrawing(e) { isDrawing = true; const rect = practiceCanvas.getBoundingClientRect(); const x = (e.clientX || e.pageX) - rect.left; const y = (e.clientY || e.pageY) - rect.top; practiceCtx.beginPath(); practiceCtx.moveTo(x, y); }
        function draw(e) { if (!isDrawing) return; const rect = practiceCanvas.getBoundingClientRect(); const x = (e.clientX || e.pageX) - rect.left; const y = (e.clientY || e.pageY) - rect.top; practiceCtx.lineTo(x, y); practiceCtx.stroke(); }
        function stopDrawing() { isDrawing = false; }
        function clearPractice() { practiceCtx.clearRect(0, 0, practiceCanvas.width, practiceCanvas.height); document.getElementById('practiceFeedback').classList.add('hidden'); }

        function displayConsonantGrid() {
            const grid = document.getElementById('consonantGrid');
            grid.innerHTML = '';
            consonants.forEach(c => {
                const card = document.createElement('div');
                card.className = 'consonant-card';
                card.innerHTML = `<div class="consonant-letter">${c.letter}</div><div class="consonant-name">${c.name}</div><div class="consonant-desc">${c.type} - ${c.direction}</div>`;
                grid.appendChild(card);
            });
        }

        function showScreen(screenId) { document.querySelectorAll('.card > div, .card > h2').forEach(el => { if (!el.classList.contains('header')) el.classList.add('hidden'); }); document.querySelectorAll('.card').forEach(card => { if (card.id !== screenId) card.classList.add('hidden'); }); document.getElementById(screenId).classList.remove('hidden'); }

        function startLesson() { document.getElementById('welcomeScreen').classList.add('hidden'); loadLesson(currentLesson); }

        function loadLesson(index) {
            if (index >= consonants.length) { showAllComplete(); return; }
            const c = consonants[index];
            showScreen('lessonScreen');
            document.getElementById('lessonTitle').textContent = `Lesson ${index + 1}: ${c.letter}`;
            document.getElementById('lessonInstruction').textContent = 'Learn this consonant stroke';
            document.getElementById('consonantDisplay').textContent = c.letter;
            document.getElementById('consonantExample').textContent = `as in "${c.example}"`;
            document.getElementById('strokeDirection').textContent = `Direction: ${c.direction} - ${c.description}`;
            document.getElementById('lessonFeedback').textContent = `💡 ${c.type === 'light' ? 'Light' : 'Heavy'} stroke - draw it ${c.type === 'light' ? 'thin' : 'thick'}.`;
        }

        function practiceStroke() { showScreen('practiceScreen'); document.getElementById('practiceConsonant').textContent = consonants[currentLesson].letter; clearPractice(); }

        function checkPractice() {
            const feedback = document.getElementById('practiceFeedback');
            feedback.classList.remove('hidden', 'correct', 'incorrect');
            feedback.classList.add('correct');
            feedback.textContent = '✓ Good! Remember the direction and weight.';
            setTimeout(() => { showQuiz(); }, 1500);
        }

        function backToLesson() { loadLesson(currentLesson); }

        function showQuiz() {
            showScreen('quizScreen');
            const c = consonants[currentLesson];
            document.getElementById('quizQuestion').textContent = `Which consonant is this? (${c.type} ${c.direction.toLowerCase()} stroke)`;
            const options = document.getElementById('quizOptions');
            options.innerHTML = '';
            let quizConsonants = [c];
            while (quizConsonants.length < 4) {
                const random = consonants[Math.floor(Math.random() * consonants.length)];
                if (!quizConsonants.includes(random)) quizConsonants.push(random);
            }
            quizConsonants.sort(() => Math.random() - 0.5);
            quizConsonants.forEach((cons, idx) => {
                const option = document.createElement('div');
                option.className = 'quiz-option';
                option.textContent = cons.letter;
                option.onclick = () => selectQuizOption(idx, cons.letter);
                options.appendChild(option);
            });
            selectedQuizOption = null;
            document.getElementById('quizFeedback').classList.add('hidden');
        }

        function selectQuizOption(index, letter) {
            document.querySelectorAll('.quiz-option').forEach((opt, idx) => { opt.classList.remove('selected'); if (idx === index) opt.classList.add('selected'); });
            selectedQuizOption = letter;
        }

        function checkQuiz() {
            if (!selectedQuizOption) { alert('Please select an answer!'); return; }
            const feedback = document.getElementById('quizFeedback');
            const currentLetter = consonants[currentLesson].letter;
            feedback.classList.remove('hidden', 'correct', 'incorrect');
            if (selectedQuizOption === currentLetter) {
                feedback.classList.add('correct');
                feedback.textContent = '✓ Correct! Well done!';
                document.querySelectorAll('.quiz-option').forEach(opt => { if (opt.textContent === currentLetter) opt.classList.add('correct'); });
                setTimeout(() => { completeLesson(); }, 1500);
            } else {
                feedback.classList.add('incorrect');
                feedback.textContent = '✗ Not quite. Try again!';
            }
        }

        function completeLesson() {
            completedLessons++;
            saveProgress();
            updateProgress();
            showScreen('completeScreen');
            document.getElementById('completeMessage').textContent = `Great job! You've mastered the ${consonants[currentLesson].letter} stroke.`;
        }

        function nextLesson() {
            currentLesson++;
            if (currentLesson >= consonants.length) { showAllComplete(); } else { loadLesson(currentLesson); }
        }

        function showAllComplete() { document.querySelectorAll('.card').forEach(card => card.classList.add('hidden')); document.getElementById('allCompleteScreen').classList.remove('hidden'); }

        function restartCourse() { currentLesson = 0; completedLessons = 0; saveProgress(); updateProgress(); document.getElementById('allCompleteScreen').classList.add('hidden'); document.getElementById('welcomeScreen').classList.remove('hidden'); displayConsonantGrid(); }

        function goHome() { alert('This would take you back to the Shorthand Academy home page!'); location.reload(); }

        function updateProgress() { const percentage = (completedLessons / consonants.length) * 100; document.getElementById('progressFill').style.width = `${percentage}%`; document.getElementById('progressText').textContent = `${completedLessons} / ${consonants.length} lessons completed`; }

        function saveProgress() { localStorage.setItem('shorthandConsonantProgress', JSON.stringify({ currentLesson, completedLessons })); }

        function loadProgress() { const saved = localStorage.getItem('shorthandConsonantProgress'); if (saved) { const data = JSON.parse(saved); currentLesson = data.currentLesson || 0; completedLessons = data.completedLessons || 0; updateProgress(); } }

        window.onload = init;
    </script>
