    <script>
        const vowels = [
            { id: 1, name: 'AH', type: 'long', symbol: '•', symbolType: 'heavy_dot', position: 1, positionName: 'First', example: 'father, calm, palm', sound: 'ah', description: 'Heavy dot in first position - written at the beginning of the stroke, above the line' },
            { id: 2, name: 'A', type: 'long', symbol: '•', symbolType: 'heavy_dot', position: 2, positionName: 'Second', example: 'plate, rate, mate', sound: 'a', description: 'Heavy dot in second position - written in the middle of the stroke, on the line' },
            { id: 3, name: 'E', type: 'long', symbol: '•', symbolType: 'heavy_dot', position: 3, positionName: 'Third', example: 'peak, ream, meet', sound: 'e', description: 'Heavy dot in third position - written at the end of the stroke' },
            { id: 4, name: 'AW', type: 'long', symbol: '—', symbolType: 'heavy_dash', position: 1, positionName: 'First', example: 'paul, rod, top', sound: 'aw', description: 'Heavy dash in first position - written at the beginning of the stroke, above the line' },
            { id: 5, name: 'O', type: 'long', symbol: '—', symbolType: 'heavy_dash', position: 2, positionName: 'Second', example: 'goat, shore, boat', sound: 'o', description: 'Heavy dash in second position - written in the middle of the stroke, on the line' },
            { id: 6, name: 'OO', type: 'long', symbol: '—', symbolType: 'heavy_dash', position: 3, positionName: 'Third', example: 'tour, shoot, boot', sound: 'oo', description: 'Heavy dash in third position - written at the end of the stroke' },
            { id: 7, name: 'a', type: 'short', symbol: '·', symbolType: 'light_dot', position: 1, positionName: 'First', example: 'bat, cat, mat', sound: 'a_short', description: 'Light dot in first position - short vowel sound' },
            { id: 8, name: 'e', type: 'short', symbol: '·', symbolType: 'light_dot', position: 2, positionName: 'Second', example: 'bet, get, met', sound: 'e_short', description: 'Light dot in second position - short vowel sound' },
            { id: 9, name: 'i', type: 'short', symbol: '·', symbolType: 'light_dot', position: 3, positionName: 'Third', example: 'bit, sit, hit', sound: 'i_short', description: 'Light dot in third position - short vowel sound' },
            { id: 10, name: 'o', type: 'short', symbol: '·', symbolType: 'light_dot', position: 1, positionName: 'First', example: 'tock, lock, mock', sound: 'o_short', description: 'Light dot in first position - short vowel sound' },
            { id: 11, name: 'u', type: 'short', symbol: '·', symbolType: 'light_dot', position: 2, positionName: 'Second', example: 'tuck, luck, much', sound: 'u_short', description: 'Light dot in second position - short vowel sound' },
            { id: 12, name: 'oo', type: 'short', symbol: '·', symbolType: 'light_dot', position: 3, positionName: 'Third', example: 'took, look, book', sound: 'oo_short', description: 'Light dot in third position - short vowel sound' }
        ];

        let currentLesson = 0;
        let completedLessons = 0;
        let selectedQuizOption = null;
        let canvas, ctx, isDrawing = false;

        function init() { canvas = document.getElementById('practiceCanvas'); ctx = canvas.getContext('2d'); setupCanvas(); loadProgress(); }

        function setupCanvas() { ctx.strokeStyle = '#667eea'; ctx.lineWidth = 3; ctx.lineCap = 'round'; ctx.lineJoin = 'round'; drawGuideLines(); canvas.addEventListener('mousedown', startDrawing); canvas.addEventListener('mousemove', draw); canvas.addEventListener('mouseup', stopDrawing); canvas.addEventListener('mouseout', stopDrawing); canvas.addEventListener('touchstart', (e) => { e.preventDefault(); startDrawing(e.touches[0]); }); canvas.addEventListener('touchmove', (e) => { e.preventDefault(); draw(e.touches[0]); }); canvas.addEventListener('touchend', stopDrawing); }

        function drawGuideLines() { ctx.clearRect(0, 0, canvas.width, canvas.height); ctx.strokeStyle = '#e0e0ff'; ctx.lineWidth = 1; ctx.beginPath(); ctx.moveTo(0, 50); ctx.lineTo(300, 50); ctx.stroke(); ctx.beginPath(); ctx.moveTo(0, 100); ctx.lineTo(300, 100); ctx.stroke(); ctx.beginPath(); ctx.moveTo(0, 150); ctx.lineTo(300, 150); ctx.stroke(); }

        function startDrawing(e) { isDrawing = true; const rect = canvas.getBoundingClientRect(); const x = (e.clientX || e.pageX) - rect.left; const y = (e.clientY || e.pageY) - rect.top; ctx.beginPath(); ctx.moveTo(x, y); }
        function draw(e) { if (!isDrawing) return; const rect = canvas.getBoundingClientRect(); const x = (e.clientX || e.pageX) - rect.left; const y = (e.clientY || e.pageY) - rect.top; ctx.lineTo(x, y); ctx.stroke(); }
        function stopDrawing() { isDrawing = false; }
        function clearPractice() { drawGuideLines(); document.getElementById('practiceFeedback').classList.add('hidden'); }

        function showScreen(screenId) { document.querySelectorAll('.card > div, .card > h2').forEach(el => { if (!el.classList.contains('header')) el.classList.add('hidden'); }); document.querySelectorAll('.card').forEach(card => { if (card.id !== screenId) card.classList.add('hidden'); }); document.getElementById(screenId).classList.remove('hidden'); }

        function startLesson() { document.getElementById('welcomeScreen').classList.add('hidden'); loadLesson(currentLesson); }

        function loadLesson(index) {
            if (index >= vowels.length) { showAllComplete(); return; }
            const vowel = vowels[index];
            showScreen('lessonScreen');
            document.getElementById('lessonTitle').textContent = `Lesson ${index + 1}: ${vowel.name}`;
            document.getElementById('lessonInstruction').textContent = 'Learn this vowel sound';
            document.getElementById('vowelSoundName').textContent = vowel.name;
            document.getElementById('vowelExample').textContent = `as in "${vowel.example}"`;
            document.getElementById('positionLabel').textContent = `${vowel.positionName} position`;
            document.getElementById('vowelType').textContent = vowel.type === 'long' ? 'Heavy ' + (vowel.symbolType.includes('dot') ? 'dot' : 'dash') : 'Light dot';
            document.getElementById('lessonFeedback').textContent = `💡 ${vowel.description}`;
            const display = document.getElementById('vowelDisplay');
            display.textContent = vowel.symbol;
            display.style.top = vowel.position === 1 ? '10px' : vowel.position === 2 ? '55px' : '100px';
            display.style.left = '125px';
        }

        function playVowelSound() { const vowel = vowels[currentLesson]; alert(`🔊 Playing sound: ${vowel.name} (as in "${vowel.example}")`); }
        function practiceVowel() { showScreen('practiceScreen'); document.getElementById('practiceVowelName').textContent = vowels[currentLesson].name; clearPractice(); }

        function checkPractice() {
            const feedback = document.getElementById('practiceFeedback');
            feedback.classList.remove('hidden', 'correct', 'incorrect');
            feedback.classList.add('correct');
            feedback.textContent = '✓ Good! Remember to place the vowel in the correct position.';
            setTimeout(() => { showQuiz(); }, 1500);
        }

        function backToLesson() { loadLesson(currentLesson); }

        function showQuiz() {
            showScreen('quizScreen');
            const vowel = vowels[currentLesson];
            document.getElementById('quizQuestion').textContent = `Which vowel is this? (${vowel.type} ${vowel.symbolType.includes('dot') ? 'dot' : 'dash'}, ${vowel.positionName} position)`;
            const options = document.getElementById('quizOptions');
            options.innerHTML = '';
            let quizVowels = [vowel];
            while (quizVowels.length < 4) {
                const random = vowels[Math.floor(Math.random() * vowels.length)];
                if (!quizVowels.includes(random)) quizVowels.push(random);
            }
            quizVowels.sort(() => Math.random() - 0.5);
            quizVowels.forEach((v, idx) => {
                const option = document.createElement('div');
                option.className = 'quiz-option';
                option.textContent = v.name;
                option.onclick = () => selectQuizOption(idx, v.id);
                options.appendChild(option);
            });
            selectedQuizOption = null;
            document.getElementById('quizFeedback').classList.add('hidden');
        }

        function selectQuizOption(index, vowelId) {
            document.querySelectorAll('.quiz-option').forEach((opt, idx) => { opt.classList.remove('selected'); if (idx === index) opt.classList.add('selected'); });
            selectedQuizOption = vowelId;
        }

        function checkQuiz() {
            if (!selectedQuizOption) { alert('Please select an answer!'); return; }
            const feedback = document.getElementById('quizFeedback');
            const currentVowelId = vowels[currentLesson].id;
            feedback.classList.remove('hidden', 'correct', 'incorrect');
            if (selectedQuizOption === currentVowelId) {
                feedback.classList.add('correct');
                feedback.textContent = '✓ Correct! Well done!';
                document.querySelectorAll('.quiz-option').forEach(opt => { if (opt.textContent === vowels[currentLesson].name) opt.classList.add('correct'); });
                setTimeout(() => { completeLesson(); }, 1500);
            } else {
                feedback.classList.add('incorrect');
                feedback.textContent = '✗ Not quite. Try again!';
            }
        }

        function playQuizSound() { const vowel = vowels[currentLesson]; alert(`🔊 Playing sound: ${vowel.name}`); }

        function completeLesson() {
            completedLessons++;
            saveProgress();
            updateProgress();
            showScreen('completeScreen');
            document.getElementById('completeMessage').textContent = `Great job! You've mastered the ${vowels[currentLesson].name} vowel.`;
        }

        function nextLesson() {
            currentLesson++;
            if (currentLesson >= vowels.length) { showAllComplete(); } else { loadLesson(currentLesson); }
        }

        function showAllComplete() { document.querySelectorAll('.card').forEach(card => card.classList.add('hidden')); document.getElementById('allCompleteScreen').classList.remove('hidden'); }

        function restartCourse() { currentLesson = 0; completedLessons = 0; saveProgress(); updateProgress(); document.getElementById('allCompleteScreen').classList.add('hidden'); document.getElementById('welcomeScreen').classList.remove('hidden'); }

        function goHome() { alert('This would take you back to the Shorthand Academy home page!'); location.reload(); }

        function updateProgress() { const percentage = (completedLessons / vowels.length) * 100; document.getElementById('progressFill').style.width = `${percentage}%`; document.getElementById('progressText').textContent = `${completedLessons} / ${vowels.length} lessons completed`; }

        function saveProgress() { localStorage.setItem('shorthandVowelProgress', JSON.stringify({ currentLesson, completedLessons })); }

        function loadProgress() { const saved = localStorage.getItem('shorthandVowelProgress'); if (saved) { const data = JSON.parse(saved); currentLesson = data.currentLesson || 0; completedLessons = data.completedLessons || 0; updateProgress(); } }

        window.onload = init;
    </script>
