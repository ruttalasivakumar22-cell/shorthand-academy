        function init() { loadAllProgress(); }
        function showModule(moduleId) { document.querySelectorAll('.nav-item').forEach(item => item.classList.remove('active')); event.currentTarget.classList.add('active'); document.querySelectorAll('.module-content').forEach(content => content.classList.remove('active')); document.getElementById(moduleId).classList.add('active'); window.scrollTo({ top: 0, behavior: 'smooth' }); }
        function launchModule(version) { const files = {'v0.3': 'shorthand_academy_pitman_lab_v0_3.html', 'v0.4': 'shorthand_academy_vowel_lab_v0_4.html', 'v0.5': 'shorthand_academy_word_builder_v0_5.html', 'v0.6': 'shorthand_academy_dictation_v0_6.html', 'v0.7': 'shorthand_academy_speed_test_v0_7.html'}; alert('🚀 Opening ' + version + '...\n\nFile: ' + files[version] + '\n\nMake sure all files are in the same folder!'); window.open(files[version], '_blank'); }
        function loadAllProgress() { let totalLessons = 0, totalWords = 0, bestWpm = 0; const vowelProgress = localStorage.getItem('shorthandVowelProgress'); if (vowelProgress) { const data = JSON.parse(vowelProgress); totalLessons += data.completedLessons || 0; } const wordProgress = localStorage.getItem('shorthandWordBuilderStats'); if (wordProgress) { const data = JSON.parse(wordProgress); totalWords += data.wordsCompleted || 0; } const speedProgress = localStorage.getItem('shorthandSpeedTestProgress'); if (speedProgress) { const data = JSON.parse(speedProgress); bestWpm = data.bestWpm || 0; } document.getElementById('totalLessons').textContent = totalLessons; document.getElementById('bestWpm').textContent = bestWpm; document.getElementById('totalWords').textContent = totalWords; document.getElementById('streakDays').textContent = 3; document.getElementById('totalLessonsStat').textContent = totalLessons; document.getElementById('bestWpmStat').textContent = bestWpm; document.getElementById('streakStat').textContent = '3 days'; }
        window.onload = init;
