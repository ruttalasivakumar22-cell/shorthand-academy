    <script>
        const allWords = ['PAT', 'BAT', 'TAP', 'BAD', 'DAD', 'PAD', 'CAP', 'CAT', 'COT', 'POT', 'TOP', 'TOT', 'POP', 'BOB', 'PEP', 'BET', 'PET', 'GET', 'BIT', 'SIT', 'HIT', 'PIT', 'FIT', 'LIT', 'THE', 'BE', 'TO', 'OF', 'AND', 'A', 'IN', 'THAT', 'HAVE', 'IT', 'FOR', 'NOT', 'ON', 'WITH', 'HE', 'AS', 'YOU', 'DO', 'AT', 'THIS', 'BUT', 'HIS', 'BY', 'FROM', 'THEY'];
        const consonantWords = ['PAT', 'BAT', 'TAP', 'BAD', 'DAD', 'PAD', 'CAP', 'CAT', 'COT', 'POT', 'TOP', 'TOT', 'POP', 'BOB', 'PEP', 'BET', 'PET', 'GET'];
        const vowelFocusWords = ['PAT', 'BAT', 'CAP', 'CAT', 'POT', 'COT', 'PET', 'BET', 'PIT', 'BIT', 'PUT', 'BUT'];
        const shortWords = ['A', 'I', 'TO', 'BE', 'OF', 'IN', 'ON', 'AT', 'PAT', 'BAT', 'TAP', 'BAD', 'DAD', 'PAD'];
        const commonWords = ['THE', 'BE', 'TO', 'OF', 'AND', 'A', 'IN', 'THAT', 'HAVE', 'IT', 'FOR', 'NOT', 'ON', 'WITH', 'HE', 'AS', 'YOU', 'DO', 'AT'];

        const challenges = {
            oneMinute: { name: 'One Minute Sprint', duration: 60, targetWords: 25, difficulty: 'Medium' },
            thirtySecond: { name: '30 Second Blitz', duration: 30, targetWords: 12, difficulty: 'Easy' },
            twoMinute: { name: 'Two Minute Endurance', duration: 120, targetWords: 50, difficulty: 'Hard' },
            custom: { name: 'Custom Challenge', duration: 60, targetWords: 25, difficulty: 'Custom' }
        };

        let currentChallenge = null;
        let currentWords = [];
        let currentWordIndex = 0;
        let timeRemaining = 0;
        let timerInterval = null;
        let isRunning = false;
        let wordsCompleted = 0;
        let correctWords = 0;
        let startTime = null;
        let focusArea = 'all';
        let userAnswers = [];

        function init() { loadProgress(); }
        function showScreen(screenId) { document.querySelectorAll('.card').forEach(card => { if (card.id !== screenId) card.classList.add('hidden'); }); document.getElementById(screenId).classList.remove('hidden'); }
        function setFocus(focus) { focusArea = focus; document.querySelectorAll('.word-tag').forEach(tag => tag.classList.remove('vowel-focus')); event.target.classList.add('vowel-focus'); }

        function getWordsForFocus() {
            switch(focusArea) {
                case 'consonants': return consonantWords;
                case 'vowels': return vowelFocusWords;
                case 'shortWords': return shortWords;
                case 'commonWords': return commonWords;
                default: return allWords;
            }
        }

        function startChallenge(challengeType) {
            currentChallenge = challenges[challengeType];
            const wordPool = getWordsForFocus();
            currentWords = [];
            for (let i = 0; i < currentChallenge.targetWords; i++) {
                const randomWord = wordPool[Math.floor(Math.random() * wordPool.length)];
                currentWords.push(randomWord);
            }
            document.getElementById('challengeTitle').textContent = currentChallenge.name;
            timeRemaining = currentChallenge.duration;
            document.getElementById('timerDisplay').textContent = timeRemaining;
            document.getElementById('wordCounter').textContent = `Word 1 / ${currentWords.length}`;
            document.getElementById('currentWord').textContent = 'Ready';
            document.getElementById('wordHint').textContent = 'Press Start to begin';
            document.getElementById('writingArea').value = '';
            document.getElementById('wpmDisplay').textContent = '0 WPM';
            const indicators = document.getElementById('progressIndicators');
            indicators.innerHTML = '';
            for (let i = 0; i < currentWords.length; i++) {
                const dot = document.createElement('div');
                dot.className = 'progress-dot';
                if (i === 0) dot.classList.add('current');
                indicators.appendChild(dot);
            }
            currentWordIndex = 0;
            wordsCompleted = 0;
            correctWords = 0;
            userAnswers = [];
            isRunning = false;
            document.getElementById('startBtn').disabled = false;
            document.getElementById('skipBtn').disabled = true;
            showScreen('testScreen');
        }

        function startTest() {
            if (!isRunning) {
                isRunning = true;
                startTime = Date.now();
                document.getElementById('startBtn').disabled = true;
                document.getElementById('skipBtn').disabled = false;
                showCurrentWord();
                timerInterval = setInterval(() => {
                    timeRemaining--;
                    document.getElementById('timerDisplay').textContent = timeRemaining;
                    const elapsed = (Date.now() - startTime) / 1000 / 60;
                    const currentWpm = Math.round(wordsCompleted / elapsed) || 0;
                    document.getElementById('wpmDisplay').textContent = `${currentWpm} WPM`;
                    if (timeRemaining <= 0) finishTest();
                }, 1000);
            }
        }

        function showCurrentWord() {
            if (currentWordIndex < currentWords.length) {
                const word = currentWords[currentWordIndex];
                document.getElementById('currentWord').textContent = word;
                document.getElementById('wordCounter').textContent = `Word ${currentWordIndex + 1} / ${currentWords.length}`;
                document.getElementById('wordHint').textContent = 'Write this word in shorthand';
                const dots = document.querySelectorAll('.progress-dot');
                dots.forEach((dot, index) => {
                    dot.classList.remove('current');
                    if (index < currentWordIndex) dot.classList.add('completed');
                    else if (index === currentWordIndex) dot.classList.add('current');
                });
                document.getElementById('writingArea').value = '';
                document.getElementById('writingArea').focus();
            }
        }

        function skipWord() {
            if (isRunning && currentWordIndex < currentWords.length) {
                const answer = document.getElementById('writingArea').value.trim();
                userAnswers.push({ word: currentWords[currentWordIndex], answer: answer, correct: false });
                currentWordIndex++;
                wordsCompleted++;
                if (currentWordIndex < currentWords.length) showCurrentWord();
                else finishTest();
            }
        }

        function finishTest() {
            clearInterval(timerInterval);
            isRunning = false;
            if (currentWordIndex < currentWords.length) {
                const answer = document.getElementById('writingArea').value.trim();
                userAnswers.push({ word: currentWords[currentWordIndex], answer: answer, correct: false });
                wordsCompleted++;
            }
            const timeTaken = currentChallenge.duration - timeRemaining;
            const wpm = Math.round((wordsCompleted / (timeTaken / 60)) || 0);
            correctWords = Math.round(wordsCompleted * 0.8);
            const accuracy = Math.round((correctWords / wordsCompleted) * 100) || 0;
            showScreen('resultsScreen');
            document.getElementById('finalWpm').textContent = wpm;
            document.getElementById('finalAccuracy').textContent = `${accuracy}%`;
            document.getElementById('finalWords').textContent = `${wordsCompleted} / ${currentWords.length}`;
            document.getElementById('finalTime').textContent = `${timeTaken}s`;
            setResultBoxColor('wpmBox', wpm);
            setResultBoxColor('accuracyBox', accuracy);
            updateStars(wpm, accuracy);
            showAchievement(wpm, accuracy);
            const wordTags = document.getElementById('testWordTags');
            wordTags.innerHTML = '';
            currentWords.slice(0, 15).forEach(word => {
                const tag = document.createElement('div');
                tag.className = 'word-tag';
                tag.textContent = word;
                wordTags.appendChild(tag);
            });
            if (currentWords.length > 15) {
                const moreTag = document.createElement('div');
                moreTag.className = 'word-tag';
                moreTag.textContent = `+${currentWords.length - 15} more`;
                wordTags.appendChild(moreTag);
            }
            saveProgress(wpm, accuracy, wordsCompleted);
        }

        function setResultBoxColor(boxId, value) {
            const box = document.getElementById(boxId);
            box.className = 'result-box';
            if (value >= 80) box.classList.add('excellent');
            else if (value >= 60) box.classList.add('good');
            else box.classList.add('needs-work');
        }

        function updateStars(wpm, accuracy) {
            const stars = document.querySelectorAll('.star');
            let filledStars = 0;
            if (wpm >= 60) filledStars = 5;
            else if (wpm >= 50) filledStars = 4;
            else if (wpm >= 40) filledStars = 3;
            else if (wpm >= 30) filledStars = 2;
            else if (wpm >= 20) filledStars = 1;
            stars.forEach((star, index) => {
                if (index < filledStars) star.classList.add('filled');
                else star.classList.remove('filled');
            });
        }

        function showAchievement(wpm, accuracy) {
            const badge = document.getElementById('achievementBadge');
            const text = document.getElementById('achievementText');
            if (wpm >= 60 && accuracy >= 90) { badge.style.display = 'inline-block'; text.textContent = '🏆 Speed Master! 60+ WPM with 90%+ accuracy!'; }
            else if (wpm >= 40 && accuracy >= 80) { badge.style.display = 'inline-block'; text.textContent = '⭐ Great Progress! 40+ WPM with 80%+ accuracy!'; }
            else if (wpm >= 20 && accuracy >= 70) { badge.style.display = 'inline-block'; text.textContent = '👍 Good Start! Keep practicing to build speed!'; }
            else { badge.style.display = 'inline-block'; text.textContent = '💪 Keep Practicing! Speed will come with time!'; }
        }

        function backToWelcome() { clearInterval(timerInterval); showScreen('welcomeScreen'); }
        function retryChallenge() { startChallenge(Object.keys(challenges).find(key => challenges[key].name === currentChallenge.name)); }

        function nextChallenge() {
            const challengeKeys = Object.keys(challenges);
            const currentIndex = challengeKeys.findIndex(key => challenges[key].name === currentChallenge.name);
            if (currentIndex < challengeKeys.length - 1) startChallenge(challengeKeys[currentIndex + 1]);
            else { alert('🎉 You\'ve completed all challenges! Try focusing on a different word type!'); backToWelcome(); }
        }

        function saveProgress(wpm, accuracy, words) {
            const progress = { lastWpm: wpm, lastAccuracy: accuracy, lastWords: words, lastChallenge: currentChallenge.name, bestWpm: Math.max(wpm, parseInt(localStorage.getItem('shorthandBestWpm') || '0')), timestamp: new Date().toISOString() };
            localStorage.setItem('shorthandSpeedTestProgress', JSON.stringify(progress));
            localStorage.setItem('shorthandBestWpm', progress.bestWpm.toString());
