    <script>
        const basicWords = ['PAT', 'BAT', 'TAP', 'BAD', 'DAD', 'PAD', 'CAP', 'CAT', 'COT', 'POT', 'TOP', 'TOT', 'POP', 'BOB', 'PEP', 'BET', 'PET', 'GET', 'BIT', 'SIT', 'HIT', 'PIT', 'FIT', 'LIT'];
        const commonWords = ['the', 'be', 'to', 'of', 'and', 'a', 'in', 'that', 'have', 'I', 'it', 'for', 'not', 'on', 'with', 'he', 'as', 'you', 'do', 'at', 'this', 'but', 'his', 'by', 'from', 'they', 'we', 'say', 'her', 'she', 'or', 'an', 'will', 'my', 'one', 'all', 'would', 'there', 'their', 'what'];
        const sentences = ['Pat the dog', 'Bat and ball', 'Tap the water', 'Bad day today', 'Dad is here', 'Pad and pen', 'Cap on head', 'Cat on mat', 'Pot on stove', 'Top of the hill'];

        let currentSpeed = 20;
        let currentExercise = 'basic';
        let currentWords = [];
        let currentWordIndex = 0;
        let isPlaying = false;
        let isPaused = false;
        let timerInterval;
        let startTime;
        let elapsedTime = 0;
        let userTranscription = [];
        let dictationStartTime;
        let dictationEndTime;

        function init() { loadProgress(); }
        function showScreen(screenId) { document.querySelectorAll('.card').forEach(card => { if (card.id !== screenId) card.classList.add('hidden'); }); document.getElementById(screenId).classList.remove('hidden'); }
        function selectSpeed(speed) { currentSpeed = speed; selectExercise('basic'); }

        function selectExercise(exercise) {
            currentExercise = exercise;
            if (exercise === 'basic') currentWords = basicWords.slice(0, 20);
            else if (exercise === 'common') currentWords = commonWords.slice(0, 20);
            else if (exercise === 'sentences') currentWords = sentences.slice(0, 10);
            currentWords.sort(() => Math.random() - 0.5);
            startDictationInterface();
        }

        function startDictationInterface() {
            showScreen('dictationScreen');
            currentWordIndex = 0;
            userTranscription = [];
            elapsedTime = 0;
            isPlaying = false;
            isPaused = false;
            document.getElementById('currentWord').textContent = 'Ready';
            document.getElementById('wordCount').textContent = `Word 0 / ${currentWords.length}`;
            document.getElementById('transcriptionBox').value = '';
            document.getElementById('timerDisplay').textContent = '00:00';
            updateSpeedButtons();
            displayWordList();
            updateProgress();
        }

        function updateSpeedButtons() {
            document.querySelectorAll('.speed-btn').forEach(btn => btn.classList.remove('active'));
            const activeBtn = document.getElementById(`speed${currentSpeed}`);
            if (activeBtn) activeBtn.classList.add('active');
        }

        function displayWordList() {
            const list = document.getElementById('wordList');
            list.innerHTML = '';
            currentWords.forEach((word, index) => {
                const item = document.createElement('div');
                item.className = 'word-item';
                if (index === currentWordIndex) item.classList.add('current');
                if (index < currentWordIndex) item.classList.add('completed');
                item.textContent = word;
                list.appendChild(item);
            });
        }

        function updateProgress() {
            const percent = (currentWordIndex / currentWords.length) * 100;
            document.getElementById('progressFill').style.width = `${percent}%`;
            document.getElementById('progressPercent').textContent = `${Math.round(percent)}%`;
            document.getElementById('wordsCompleted').textContent = `${currentWordIndex} / ${currentWords.length} words`;
        }

        function startDictation() {
            if (!isPlaying) {
                isPlaying = true;
                isPaused = false;
                document.getElementById('playBtn').disabled = true;
                document.getElementById('pauseBtn').disabled = false;
                document.getElementById('stopBtn').disabled = false;
                document.getElementById('replayBtn').disabled = false;
                if (elapsedTime === 0) {
                    dictationStartTime = new Date();
                    startTime = Date.now();
                    startTimer();
                } else {
                    startTime = Date.now() - elapsedTime * 1000;
                }
                playCurrentWord();
            }
        }

        function pauseDictation() {
            if (isPlaying && !isPaused) {
                isPaused = true;
                elapsedTime = (Date.now() - startTime) / 1000;
                document.getElementById('playBtn').disabled = false;
                document.getElementById('pauseBtn').disabled = true;
                clearInterval(timerInterval);
            } else if (isPaused) {
                isPaused = false;
                startTime = Date.now() - elapsedTime * 1000;
                document.getElementById('playBtn').disabled = true;
                document.getElementById('pauseBtn').disabled = false;
                startTimer();
            }
        }

        function stopDictation() {
            isPlaying = false;
            isPaused = false;
            clearInterval(timerInterval);
            document.getElementById('playBtn').disabled = false;
            document.getElementById('pauseBtn').disabled = true;
            document.getElementById('stopBtn').disabled = true;
            document.getElementById('replayBtn').disabled = true;
            document.getElementById('currentWord').textContent = 'Stopped';
        }

        function replayWord() {
            if (currentWordIndex > 0) {
                currentWordIndex--;
                updateProgress();
                displayWordList();
                playCurrentWord();
            }
        }

        function playCurrentWord() {
            if (currentWordIndex < currentWords.length) {
                const word = currentWords[currentWordIndex];
                document.getElementById('currentWord').textContent = word;
                document.getElementById('wordCount').textContent = `Word ${currentWordIndex + 1} / ${currentWords.length}`;
                const wordDuration = 60 / currentSpeed;
                setTimeout(() => {
                    if (isPlaying && !isPaused) {
                        currentWordIndex++;
                        updateProgress();
                        displayWordList();
                        if (currentWordIndex < currentWords.length) playCurrentWord();
                        else finishDictation();
                    }
                }, wordDuration * 1000);
            }
        }

        function startTimer() {
            timerInterval = setInterval(() => {
                if (!isPaused) {
                    elapsedTime = (Date.now() - startTime) / 1000;
                    const minutes = Math.floor(elapsedTime / 60);
                    const seconds = Math.floor(elapsedTime % 60);
                    document.getElementById('timerDisplay').textContent = `${minutes.toString().padStart(2, '0')}:${seconds.toString().padStart(2, '0')}`;
                }
            }, 1000);
        }

        function changeSpeed(speed) {
            currentSpeed = speed;
            updateSpeedButtons();
            currentWordIndex = 0;
            userTranscription = [];
            elapsedTime = 0;
            isPlaying = false;
            isPaused = false;
            document.getElementById('currentWord').textContent = 'Ready';
            document.getElementById('wordCount').textContent = `Word 0 / ${currentWords.length}`;
            document.getElementById('transcriptionBox').value = '';
            document.getElementById('timerDisplay').textContent = '00:00';
            document.getElementById('playBtn').disabled = false;
            document.getElementById('pauseBtn').disabled = true;
            document.getElementById('stopBtn').disabled = true;
            document.getElementById('replayBtn').disabled = true;
            displayWordList();
            updateProgress();
        }

        function finishDictation() {
            stopDictation();
            dictationEndTime = new Date();
            const transcription = document.getElementById('transcriptionBox').value.trim();
            const userWords = transcription.split(/\s+/).filter(w => w.length > 0);
            let correctCount = 0;
            const correctWords = [];
            const wrongWords = [];
            currentWords.forEach((word, index) => {
                const userWord = userWords[index] || '';
                if (userWord.toUpperCase() === word.toUpperCase()) {
                    correctCount++;
                    correctWords.push(word);
                } else {
                    wrongWords.push({ correct: word, user: userWord || '(missed)' });
                }
            });
            const accuracy = Math.round((correctCount / currentWords.length) * 100);
            const timeTaken = (dictationEndTime - dictationStartTime) / 1000 / 60;
            const actualWPM = Math.round(currentWords.length / timeTaken);
            showScreen('resultsScreen');
            document.getElementById('accuracyValue').textContent = `${accuracy}%`;
            document.getElementById('wpmValue').textContent = actualWPM;
            document.getElementById('wordsValue').textContent = `${correctCount} / ${currentWords.length}`;
            document.getElementById('timeValue').textContent = document.getElementById('timerDisplay').textContent;
            const accuracyCard = document.getElementById('accuracyCard');
            accuracyCard.className = 'result-card';
            if (accuracy >= 90) accuracyCard.classList.add('excellent');
            else if (accuracy >= 70) accuracyCard.classList.add('good');
            else accuracyCard.classList.add('needs-work');
            const feedback = document.getElementById('resultFeedback');
            if (accuracy >= 90) { feedback.className = 'feedback correct'; feedback.textContent = '🎉 Excellent! You\'re ready for the next speed level!'; }
            else if (accuracy >= 70) { feedback.className = 'feedback info'; feedback.textContent = '👍 Good job! A bit more practice and you\'ll master this speed.'; }
            else { feedback.className = 'feedback incorrect'; feedback.textContent = '💪 Keep practicing! Try this speed again or go slower to build confidence.'; }
            const correctList = document.getElementById('correctWordsList');
            correctList.innerHTML = '';
            correctWords.forEach(word => { const item = document.createElement('div'); item.className = 'word-item completed'; item.textContent = word; correctList.appendChild(item); });
            const reviewList = document.getElementById('reviewWordsList');
            reviewList.innerHTML = '';
            wrongWords.forEach(w => { const item = document.createElement('div'); item.className = 'word-item'; item.style.borderColor = '#dc3545'; item.style.background = '#f8d7da'; item.innerHTML = `<strong>${w.correct}</strong><br><small>You: ${w.user}</small>`; reviewList.appendChild(item); });
            saveProgress(accuracy, actualWPM);
        }

        function backToWelcome() { showScreen('welcomeScreen'); }
        function retryDictation() { currentWordIndex = 0; userTranscription = []; elapsedTime = 0; isPlaying = false; isPaused = false; startDictationInterface(); }

        function nextLevel() {
            const speeds = [20, 30, 40, 50, 60, 80, 100, 120];
            const currentIndex = speeds.indexOf(currentSpeed);
            if (currentIndex < speeds.length - 1) {
                currentSpeed = speeds[currentIndex + 1];
                selectExercise(currentExercise);
            } else {
                alert('🏆 Congratulations! You\'ve reached the maximum speed level!');
                backToWelcome();
            }
        }

        function saveProgress(accuracy, wpm) {
            const progress = { lastSpeed: currentSpeed, lastAccuracy: accuracy, lastWPM: wpm, lastExercise: currentExercise, timestamp: new Date().toISOString() };
            localStorage.setItem('shorthandDictationProgress', JSON.stringify(progress));
        }

        function loadProgress() {
            const saved = localStorage.getItem('shorthandDictationProgress');
            if (saved) {
                const progress = JSON.parse(saved);
                if (progress.lastSpeed) currentSpeed = progress.lastSpeed;
                if (progress.lastExercise) currentExercise = progress.lastExercise;
            }
        }

        window.onload = init;
    </script>
